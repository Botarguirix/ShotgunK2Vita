  set(SG_PATCH_OPTIMIZED "" CACHE FILEPATH "Private optimized script")
  set(SG_PATCH_TARGET "" CACHE FILEPATH "Private target script")
  set(SG_PATCH_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/code_patches.h")
  if(SG_PATCH_ORIGINAL OR SG_PATCH_OPTIMIZED OR SG_PATCH_TARGET)
    if(NOT SG_PATCH_ORIGINAL OR NOT SG_PATCH_OPTIMIZED OR NOT SG_PATCH_TARGET)
      message(FATAL_ERROR "Specify all three private patch inputs, or none")
    endif()
    set(SG_PATCH_ARGS "${SG_PATCH_ORIGINAL}" "${SG_PATCH_OPTIMIZED}" "${SG_PATCH_TARGET}")
    set(SG_PATCH_DEPS "${SG_PATCH_ORIGINAL}" "${SG_PATCH_OPTIMIZED}" "${SG_PATCH_TARGET}")
  else()
    set(SG_PATCH_ARGS --empty)
  endif()
  add_custom_command(OUTPUT "${SG_PATCH_HEADER}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_code_patches.py" ${SG_PATCH_ARGS} "${SG_PATCH_HEADER}"
    DEPENDS tools/gen_code_patches.py ${SG_PATCH_DEPS} VERBATIM)

    "static const unsigned char sg_steam_compat_lua[]={${SG_COMPAT_BYTES}};\n")
