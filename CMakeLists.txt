cmake_minimum_required(VERSION 3.16)
option(SG_HOST_PROBE "Build a desktop Lua bootstrap probe" OFF)
if(SG_HOST_PROBE)
  project(ShotgunK2VitaProbe C)
else()
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(NOT DEFINED ENV{VITASDK})
    message(FATAL_ERROR "Set VITASDK to your VitaSDK installation")
  endif()
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(ShotgunK2Vita C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
endif()
file(GLOB LUA_SOURCES third_party/lua-5.1.5/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|print)\\.c$")
add_library(sg_lua STATIC ${LUA_SOURCES})
target_include_directories(sg_lua PUBLIC third_party/lua-5.1.5/src)
add_library(sg_runtime STATIC src/runtime.c src/graphics.c src/perf.c src/render_pool.c src/audio.c src/controls.c src/diagnostics.c)
target_compile_options(sg_lua PRIVATE -g)
target_compile_options(sg_runtime PRIVATE -g)
target_include_directories(sg_runtime PRIVATE third_party)
target_link_libraries(sg_runtime PUBLIC sg_lua m)
target_include_directories(sg_runtime PUBLIC src)
if(SG_HOST_PROBE)
  find_package(Threads REQUIRED)
  target_link_libraries(sg_runtime PUBLIC Threads::Threads)
  add_executable(sg_probe src/host_probe.c)
  target_link_libraries(sg_probe sg_runtime)
  add_executable(sg_bench src/host_bench.c)
  target_link_libraries(sg_bench sg_runtime)
  add_executable(sg_audio_probe src/host_audio_probe.c)
  target_link_libraries(sg_audio_probe sg_runtime)
  add_executable(sg_inputs_probe tests/test_inputs.c)
  target_link_libraries(sg_inputs_probe sg_runtime)
  target_compile_options(sg_inputs_probe PRIVATE -UNDEBUG)
  add_executable(sg_diagnostics_probe tests/test_diagnostics.c)
  target_link_libraries(sg_diagnostics_probe sg_runtime)
  target_compile_options(sg_diagnostics_probe PRIVATE -UNDEBUG)
else()
target_compile_definitions(sg_runtime PUBLIC SG_VITA=1)
add_executable(ShotgunK2Vita src/main.c)
target_compile_options(ShotgunK2Vita PRIVATE -Wall -Wextra -g)
target_link_options(ShotgunK2Vita PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/ShotgunK2Vita.map")
target_include_directories(ShotgunK2Vita PRIVATE third_party)
target_link_libraries(ShotgunK2Vita sg_runtime SceDisplay_stub SceCtrl_stub SceTouch_stub SceSysmem_stub SceKernelThreadMgr_stub SceAudio_stub ScePower_stub)
vita_create_self(ShotgunK2Vita.self ShotgunK2Vita)
set(VITA_LIVEAREA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/private/livearea" CACHE PATH "Locally extracted game artwork")
set(VITA_LIVEAREA_FILES)
if(EXISTS "${VITA_LIVEAREA_DIR}/livearea/contents/template.xml")
  foreach(ASSET icon0.png pic0.png livearea/contents/bg0.png livearea/contents/startup.png livearea/contents/template.xml)
    if(NOT EXISTS "${VITA_LIVEAREA_DIR}/${ASSET}")
      message(FATAL_ERROR "Missing LiveArea file: ${ASSET}")
    endif()
    list(APPEND VITA_LIVEAREA_FILES FILE "${VITA_LIVEAREA_DIR}/${ASSET}" "sce_sys/${ASSET}")
  endforeach()
endif()
vita_create_vpk(ShotgunK2Vita-performance.vpk SGKV00001 ShotgunK2Vita.self
  VERSION "00.05" NAME "ShotgunK2Vita" ${VITA_LIVEAREA_FILES})
endif()
