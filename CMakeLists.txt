cmake_minimum_required(VERSION 3.16)
option(SG_HOST_PROBE "Build a desktop Lua bootstrap probe" OFF)
if(SG_HOST_PROBE)
  project(ShotgunK2VitaProbe C)
else()
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(NOT DEFINED ENV{VITASDK})
    message(FATAL_ERROR "Set VITASDK to your VitaSDK installation")
  endif()
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(ShotgunK2Vita C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
endif()
file(GLOB LUA_SOURCES third_party/lua-5.1.5/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|print)\\.c$")
add_library(sg_lua STATIC ${LUA_SOURCES})
target_include_directories(sg_lua PUBLIC third_party/lua-5.1.5/src)
add_library(sg_runtime STATIC src/runtime.c src/graphics.c)
target_include_directories(sg_runtime PRIVATE third_party)
target_link_libraries(sg_runtime PUBLIC sg_lua m)
target_include_directories(sg_runtime PUBLIC src)
if(SG_HOST_PROBE)
  add_executable(sg_probe src/host_probe.c)
  target_link_libraries(sg_probe sg_runtime)
else()
add_executable(ShotgunK2Vita src/main.c)
target_compile_options(ShotgunK2Vita PRIVATE -Wall -Wextra)
target_include_directories(ShotgunK2Vita PRIVATE third_party)
target_link_libraries(ShotgunK2Vita sg_runtime SceDisplay_stub SceCtrl_stub SceTouch_stub SceSysmem_stub)
vita_create_self(ShotgunK2Vita.self ShotgunK2Vita)
vita_create_vpk(ShotgunK2Vita-bootstrap.vpk SGKV00001 ShotgunK2Vita.self
  VERSION "00.02" NAME "ShotgunK2Vita Bootstrap")
endif()
